        ::= _

              ::= $

            ::= Expr Index
            ::= Expr [ Expr .. Expr]

            ::= transpose ( Expr ; integernum ; integernum )
            ::= Expr ^ Expr
            ::= [ List ]

          ::= ( restart Operator every Expr )
          ::= ( Iterator_mw Operator << Expr >> if Expr default Expr )
          ::= ( foldw Operator << Expr >> if Expr )
          ::= ( foldwi Operator << Expr >> if Expr )
