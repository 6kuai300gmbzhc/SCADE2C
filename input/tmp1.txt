package airdynamic

function sum_pmmmm ( x1 : ' t ;
x2 : ' t ;
x3 : ' t ;
x4 : ' t ;
x5 : ' t ) returns ( y : ' t ) where ' t numeric 
let
y = x1 - x2 - x3 - x4 - x5 ;
tel

function Subtract ( In1 : float64 ;
In2 : float64 ;
In3 : float64 ;
In4 : float64 ;
In5 : float64 ) returns ( Out1 : float64 )
let
    Out1 = sum_pmmmm ( In1 , In2 , In3 , In4 , In5 ) ;
tel

end;

package pbtb

  const k2 : float64 = Tsm / Px;

  const P0 : float64 = 0.0;

  const Pm : float64 = 21000000.0;

  const Px : float64 = 19000000.0;

  const Tsm : float64 = 20000;

  node pbtb(Pb : float64; t : float64)

        _L5 : float64;
        _L4 : float64;
        _L2 : float64;
        _L9 : float64;

        _L2= P0;

        _L6= _L4 #2 - _L2;

      tel

      then let
        Tb= last 'Tb;
      tel
      else if last 'Tb #26 / k1 #13 + P0 #18 < Pb #16 and Pm #10 >= Pb

        Tb= k1 #1 * (Pb #1 - P0);

        Tb= last 'Tb;

    _L13= Pb;
    Kb= _L14;
    _L14= _L15 #1 / _L16;

    _L16= 21;
  tel
